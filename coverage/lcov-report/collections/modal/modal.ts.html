import { LitElement, unsafeCSS } from 'lit';
import { property } from 'lit/decorators.js';
import scrollStyles from '../../shared/scrollbar.style.css?inline';
import { booleanType } from '../../utils/property-types';
import modalStyles from './modal.style.css?inline';
import { modalTemplate } from './modal.template';
 
const ANIMATION_DURATION = 250;
 
export class Modal extends LitElement {
	static get styles() {
		return [unsafeCSS(scrollStyles), unsafeCSS(modalStyles)];

	@property(booleanType) open = false;
 
	@property({ type: String }) modalTitle = '';

		this.status = state;

	get _headerSlot() {
		const slot: HTMLSlotElement | undefined = this.shadowRoot?.querySelector('slot[name="header"]') || undefined;
		// this.requestUpdate();
		return slot?.assignedElements()[0] as HTMLInputElement;
	}

		const slot: HTMLSlotElement | undefined = this.shadowRoot?.querySelector('slot[name="footer"]') || undefined;
		// this.requestUpdate();
		return slot?.assignedElements()[0] as HTMLInputElement;
	}
 
	constructor() {
		super();
		this._handleKeydown = this._handleKeydown.bind(this);
		this._handleOutsideClick = this._handleOutsideClick.bind(this);
		this._scrollHandler = this._handleScroll.bind(this);
		this._handleStartSentinelFocus = this._handleStartSentinelFocus.bind(this);
		this._handleStartSentinelKeydown = this._handleStartSentinelKeydown.bind(this);
		this._handleEndSentinelFocus = this._handleEndSentinelFocus.bind(this);
	}
 
	connectedCallback() {
		super.connectedCallback();
		document.addEventListener('mousedown', this._handleOutsideClick);
	}
 
	disconnectedCallback() {
		super.disconnectedCallback();
		document.removeEventListener('mousedown', this._handleOutsideClick);
		if (this._scrollContainer && this._scrollHandler) {
			this._scrollContainer.removeEventListener('scroll', this._scrollHandler);
		}
		this._scrollContainer = null;
	}
 
	updated(changedProperties: any) {
		if (changedProperties.has('open')) {
			if (this.open) {
				this._showDialog();
			} else {
				this._hideDialog();
			}
		}
	}
 
	_showDialog() {
		this._previouslyFocusedElement = (document.activeElement as HTMLElement) || null;
		this._closedViaKeyboard = false;
		this.classList.add('show');
		this.classList.remove('hide');
		setTimeout(() => {
			this.classList.add('show');
			this.style.visibility = 'visible';
		}, 0);
		setTimeout(() => {
			const startSentinel = this.shadowRoot?.querySelector<HTMLElement>('.modal-focus-sentinel--start');
			startSentinel?.focus();
		}, 100);
		document.addEventListener('keydown', this._handleKeydown);
		document.body.style.overflow = 'hidden';
	}
 
	_hideDialog() {
		this.classList.add('hide');
		this.classList.remove('show');
		setTimeout(() => {
			this.classList.remove('hide');
			this.style.visibility = 'hidden';
			document.removeEventListener('keydown', this._handleKeydown);
			if (this._previouslyFocusedElement && typeof this._previouslyFocusedElement.focus === 'function') {
				if (this._closedViaKeyboard) {
					this._previouslyFocusedElement.focus();
				} else {
					const focusOptions = { focusVisible: false } as FocusOptions;
					this._previouslyFocusedElement.focus(focusOptions);
				}
				this._previouslyFocusedElement = null;
			}
		}, ANIMATION_DURATION);
		document.body.style.overflow = '';
	}
 
	_close(event?: Event) {
		if (event) {
			if (event instanceof KeyboardEvent) {
				this._closedViaKeyboard = true;
			} else if (event instanceof MouseEvent) {
				const isKeyboardClick = event.detail === 0 && (event as PointerEvent).pointerType !== 'mouse';
				this._closedViaKeyboard = isKeyboardClick;
			}
		}
		this.open = false;
		this._hideDialog();
		this.requestUpdate();
		const eventClose = new Event('close');
		this.dispatchEvent(eventClose);
	}
 
	_isStatusValid() {
		return this.status === 'danger' || this.status === 'error' || this.status === 'warning' || this.status === 'alert';
	}
 
	_getStatusIcon() {
		let icon = '';
		switch (this.status) {
			case 'danger':
				icon = 'warning';
				break;
			case 'error':
				icon = 'warning';
				break;
			case 'warning':
				icon = 'error';
				break;
			case 'alert':
				icon = 'error';
				break;
			default:
				icon = '';
		}
		return icon;
	}
 
	_handleStartSentinelFocus(event: FocusEvent) {
		const relatedTarget = event.relatedTarget as HTMLElement | null;
		const closeButton = this.shadowRoot?.querySelector<HTMLElement>('.dss-dialog-header-close');

		if (closeButton && (relatedTarget === closeButton || closeButton.contains(relatedTarget))) {
			closeButton.focus();
		}
	}
 
	_handleStartSentinelKeydown(event: KeyboardEvent) {
		if (event.key === 'Tab' && event.shiftKey) {
			event.preventDefault();
			if (!this.hideCloseIcon) {
				const closeButton = this.shadowRoot?.querySelector<HTMLElement>('.dss-dialog-header-close');
				closeButton?.focus();
			} else {
				const endSentinel = this.shadowRoot?.querySelector<HTMLElement>('.modal-focus-sentinel--end');
				endSentinel?.focus();
			}
		}
	}
 
	_handleEndSentinelFocus() {
		if (!this.hideCloseIcon) {
			const closeButton = this.shadowRoot?.querySelector<HTMLElement>('.dss-dialog-header-close');
			closeButton?.focus();
		} else {
			const startSentinel = this.shadowRoot?.querySelector<HTMLElement>('.modal-focus-sentinel--start');
			startSentinel?.focus();
		}
	}
 
	_handleKeydown(event: KeyboardEvent) {
		if (event.key === 'Escape') {
			this._close(event);
		}
	}
 
	_handleOutsideClick(event: any) {
		if (this.open) {
			const dialog = this.shadowRoot?.querySelector('.dss-dialog');
			const path = event.composedPath();
			if (dialog && path.includes(this) && !path.includes(dialog)) {
				this._close(event);
			}
		}
	}
 
	private fixEmptyFooter() {
		const dialog = this.shadowRoot?.querySelector('.dss-dialog');
		if (dialog?.classList.contains('dss-dialog--footer-empty')) {
			this.requestUpdate();
		}
	}
 
	private _handleScroll(event: Event) {
		const wrapper = event.target as HTMLElement;
		if (wrapper) {
			// Handle Header
			if (wrapper.scrollTop > 0) {
				this._modalHeader?.classList.add('dss-dialog-header--scrolled');
			} else {
				this._modalHeader?.classList.remove('dss-dialog-header--scrolled');
			}
			// Handle Footer
			if (wrapper.scrollHeight - wrapper.scrollTop !== wrapper.clientHeight) {
				this._modalFooter?.classList.add('dss-dialog-footer--scrolled');
			} else {
				this._modalFooter?.classList.remove('dss-dialog-footer--scrolled');
			}
		}
	}
 
	private _modalHeader: HTMLElement | null = null;
 
	private _modalFooter: HTMLElement | null = null;
 
	async firstUpdated() {
		await this.updateComplete;
		if (this._footerSlot) {

		const bodyWrapper = this.shadowRoot?.querySelector('.dss-dialog-body');

			const footerHeight = this._modalFooter ? this._modalFooter.clientHeight : 0;
			const dialogHeight = this.shadowRoot?.querySelector('.dss-dialog')?.clientHeight || 0;
			const availableHeight = dialogHeight - headerHeight - footerHeight;

			// if (bodyWrapper.scrollHeight > bodyWrapper.clientHeight) {
			if (bodyWrapper.scrollHeight > availableHeight) {
				this._modalFooter.classList.add('dss-dialog-footer--scrolled');

		}
	}
