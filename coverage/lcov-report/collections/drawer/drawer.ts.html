import { LitElement, unsafeCSS } from 'lit';
import { property } from 'lit/decorators.js';
import resetStyles from '../../shared/reset.style.css?inline';
import { booleanType } from '../../utils/property-types';
import drawerStyles from './drawer.style.css?inline';
import { template } from './drawer.template';
 
const ANIMATION_DURATION = 250;
 
export class Drawer extends LitElement {
	static get styles() {
		return [unsafeCSS(resetStyles), unsafeCSS(drawerStyles)];
	}
 
	/* PROPS */
	@property({ type: String }) variant = 'default';
 
	@property(booleanType) jcef = false;
 
	@property(booleanType) open = false;
 
	@property({ type: String }) title = '';
 
	@property({ type: String }) titleText = '';
 
	_drawerHeader: HTMLElement | null = null;
 
	_drawerFooter: HTMLElement | null = null;
 
	_scrollHandler: ((event: Event) => void) | null = null;
 
	_scrollContainer: HTMLElement | null = null;
 
	_previouslyFocusedElement: HTMLElement | null = null;
 
	_closedViaKeyboard = false;
 
	/* METHODS */
	constructor() {
		super();
		this._handleKeydown = this._handleKeydown.bind(this);
		this._handleOutsideClick = this._handleOutsideClick.bind(this);
		this._scrollHandler = this._handleScroll.bind(this);
		this._handleStartSentinelFocus = this._handleStartSentinelFocus.bind(this);
		this._handleStartSentinelKeydown = this._handleStartSentinelKeydown.bind(this);
		this._handleEndSentinelFocus = this._handleEndSentinelFocus.bind(this);
	}
 
	_showDrawer() {
		this._previouslyFocusedElement = (document.activeElement as HTMLElement) || null;
		this._closedViaKeyboard = false;
		this.classList.add('show');
		this.classList.remove('hide');
		document.addEventListener('keydown', this._handleKeydown);
		setTimeout(() => {
			this.classList.add('show');
			this.style.visibility = 'visible';
		}, 0);
		setTimeout(() => {
			const startSentinel = this.shadowRoot?.querySelector<HTMLElement>('.drawer-focus-sentinel--start');
			startSentinel?.focus();
		}, 100);
		document.body.style.overflow = 'hidden';
	}
 
	_hideDrawer() {
		this.classList.add('hide');
		this.classList.remove('show');
		document.removeEventListener('keydown', this._handleKeydown);
		setTimeout(() => {
			this.classList.remove('hide');
			this.style.visibility = 'hidden';
			if (this._previouslyFocusedElement && typeof this._previouslyFocusedElement.focus === 'function') {
				if (this._closedViaKeyboard) {
					this._previouslyFocusedElement.focus();
				} else {
					const focusOptions = { focusVisible: false } as FocusOptions;
					this._previouslyFocusedElement.focus(focusOptions);
				}
				this._previouslyFocusedElement = null;
			}
		}, ANIMATION_DURATION);
		document.body.style.overflow = '';
	}
 
	_handleClose(event?: Event) {
		if (event) {
			if (event instanceof KeyboardEvent) {
				this._closedViaKeyboard = true;
			} else if (event instanceof MouseEvent) {
				const isKeyboardClick = event.detail === 0 && (event as PointerEvent).pointerType !== 'mouse';
				this._closedViaKeyboard = isKeyboardClick;
			}
		}
		this.open = false;
		this._hideDrawer();
		this.requestUpdate();
		const eventClose = new Event('close');
		this.dispatchEvent(eventClose);
	}
 
	_handleKeydown(event: KeyboardEvent) {
		if (event.key === 'Escape') {
			this._handleClose(event);
		}
	}
 
	_handleOutsideClick(event: any) {
		if (this.open) {
			const drawer = this.shadowRoot?.querySelector('.drawer');
			const path = event.composedPath();
			if (drawer && path.includes(this) && !path.includes(drawer)) {
				this._handleClose(event);
			}
		}
	}
 
	_handleScroll(event: Event) {
		const drawer = event.target as HTMLElement;
		if (drawer) {
			// Handle Header
			if (drawer.scrollTop > 0) {
				this._drawerHeader?.classList.add('drawer-header--scrolled');
			} else {
				this._drawerHeader?.classList.remove('drawer-header--scrolled');
			}
			// Handle Footer
			if (drawer.scrollHeight - drawer.scrollTop !== drawer.clientHeight) {
				this._drawerFooter?.classList.add('drawer-footer--scrolled');
			} else {
				this._drawerFooter?.classList.remove('drawer-footer--scrolled');
			}
		}
	}
 
	_handleStartSentinelFocus(event: FocusEvent) {
		const relatedTarget = event.relatedTarget as HTMLElement | null;
		const closeButton = this.shadowRoot?.querySelector<HTMLElement>('.dss-dialog-header-close');

		if (closeButton && (relatedTarget === closeButton || closeButton.contains(relatedTarget))) {
			closeButton.focus();
		}
	}
 
	_handleStartSentinelKeydown(event: KeyboardEvent) {
		if (event.key === 'Tab' && event.shiftKey) {
			event.preventDefault();
			const closeButton = this.shadowRoot?.querySelector<HTMLElement>('.dss-dialog-header-close');
			closeButton?.focus();
		}
	}
 
	_handleEndSentinelFocus() {
		const closeButton = this.shadowRoot?.querySelector<HTMLElement>('.drawer-header-close');
		closeButton?.focus();
	}
 
	/* LIT LIFECYCLE */
	connectedCallback() {
		super.connectedCallback();
		document.addEventListener('mousedown', this._handleOutsideClick);
	}
 
	disconnectedCallback() {
		super.disconnectedCallback();
		document.removeEventListener('mousedown', this._handleOutsideClick);
		if (this._scrollContainer && this._scrollHandler) {
			this._scrollContainer.removeEventListener('scroll', this._scrollHandler);
		}
		this._scrollContainer = null;
	}
 
	updated(changedProperties: any) {
		if (changedProperties.has('open')) {
			if (this.open) {
				this._showDrawer();
			} else {
				this._hideDrawer();
			}
		}
		if (changedProperties.has('title')) {
			queueMicrotask(() => {
				if (this.title === '') return;
				this.titleText = this.title;
			});
		}
	}
 
	firstUpdated() {
		const drawer = this.shadowRoot?.querySelector('.drawer');
		if (drawer) {
			this._scrollContainer = drawer as HTMLElement;
			if (this._scrollHandler) {
				this._scrollContainer.addEventListener('scroll', this._scrollHandler);
			}
			this._drawerHeader = this.shadowRoot?.querySelector('.drawer-header') as HTMLElement;
			this._drawerFooter = this.shadowRoot?.querySelector('.drawer-footer') as HTMLElement;
 
			if (drawer.scrollHeight > drawer.clientHeight) {
				this._drawerFooter.classList.add('drawer-footer--scrolled');
			}
		}
	}
 
	render() {
		return template(this);
	}
}
